<body>


  <div id="game-container">

    <canvas id="gameCanvas" width="600" height="800"></canvas>

    <div id="ui-overlay">

      <div class="hud">

        <div>Wave: <span id="hud-wave">1</span></div>

        <div>Score: <span id="hud-score">0</span></div>

      </div>

    </div>

  </div>


  <script>

    // ==========================================

    // 1. YOUR CUSTOM TEXT / WORDS HERE

    // ==========================================

    const CUSTOM_WORDS = [

      "space", "laser", "galaxy", "stellar", "nebula", 

      "comet", "meteor", "orbit", "planet", "plasma",

      "velocity", "quantum", "gravity", "cosmos", "horizon",

      "vortex", "eclipse", "supernova", "asteroid", "pulsar"

    ];


    // ==========================================

    // 2. ENGINE SETUP

    // ==========================================

    const canvas = document.getElementById("gameCanvas");

    const ctx = canvas.getContext("2d");

    const hudScore = document.getElementById("hud-score");

    const hudWave = document.getElementById("hud-wave");


    const WIDTH = canvas.width;

    const HEIGHT = canvas.height;


    let score = 0;

    let wave = 1;

    let enemies = [];

    let lasers = [];

    let particles = [];

    let activeTarget = null;

    let gameState = "START"; // "START", "PLAYING", "GAMEOVER"


    const player = {

      x: WIDTH / 2,

      y: HEIGHT - 60,

      targetAngle: -Math.PI / 2

    };


    // ==========================================

    // 3. GAME CLASSES

    // ==========================================

    class Enemy {

      constructor(word, speed) {

        this.word = word;

        this.typed = 0;

        this.x = 80 + Math.random() * (WIDTH - 160);

        this.y = -20;

        this.speed = speed;

        this.radius = 14;

      }


      update() {

        this.y += this.speed;

        if (this.y >= player.y - 15) {

          gameOver();

        }

      }


      draw() {

        // Enemy ship hull

        ctx.save();

        ctx.translate(this.x, this.y);

        ctx.fillStyle = this === activeTarget ? "#f43f5e" : "#38bdf8";

        ctx.beginPath();

        ctx.moveTo(0, 10);

        ctx.lineTo(-8, -6);

        ctx.lineTo(8, -6);

        ctx.closePath();

        ctx.fill();

        ctx.restore();


        // Render text label

        ctx.font = "bold 15px monospace";

        ctx.textAlign = "center";


        const textY = this.y - 14;

        const totalWidth = ctx.measureText(this.word).width;

        let startX = this.x - totalWidth / 2;


        for (let i = 0; i < this.word.length; i++) {

          const char = this.word[i];

          const charWidth = ctx.measureText(char).width;

          ctx.fillStyle = i < this.typed ? "#4ade80" : "#ffffff";

          ctx.fillText(char, startX + charWidth / 2, textY);

          startX += charWidth;

        }

      }

    }


    class Laser {

      constructor(x1, y1, x2, y2) {

        this.x1 = x1;

        this.y1 = y1;

        this.x2 = x2;

        this.y2 = y2;

        this.life = 6;

      }

      update() { this.life--; }

      draw() {

        ctx.save();

        ctx.strokeStyle = `rgba(56, 189, 248, ${this.life / 6})`;

        ctx.lineWidth = 3;

        ctx.beginPath();

        ctx.moveTo(this.x1, this.y1);

        ctx.lineTo(this.x2, this.y2);

        ctx.stroke();

        ctx.restore();

      }

    }


    class Particle {

      constructor(x, y, color) {

        this.x = x;

        this.y = y;

        this.vx = (Math.random() - 0.5) * 5;

        this.vy = (Math.random() - 0.5) * 5;

        this.color = color;

        this.life = 20 + Math.random() * 10;

        this.maxLife = this.life;

      }

      update() {

        this.x += this.vx;

        this.y += this.vy;

        this.life--;

      }

      draw() {

        ctx.save();

        ctx.fillStyle = this.color;

        ctx.globalAlpha = Math.max(0, this.life / this.maxLife);

        ctx.fillRect(this.x, this.y, 2, 2);

        ctx.restore();

      }

    }


    // ==========================================

    // 4. LOGIC & SPAWNING

    // ==========================================

    function startWave(waveNumber) {

      hudWave.textContent = waveNumber;

      const count = 4 + waveNumber * 2;

      const baseSpeed = 0.4 + waveNumber * 0.1;


      for (let i = 0; i < count; i++) {

        const word = CUSTOM_WORDS[Math.floor(Math.random() * CUSTOM_WORDS.length)];

        const enemy = new Enemy(word, baseSpeed + Math.random() * 0.3);

        // Stagger spawn positions vertically

        enemy.y = -40 - (i * 65);

        enemies.push(enemy);

      }

    }


    function createExplosion(x, y, color = "#38bdf8") {

      for (let i = 0; i < 24; i++) {

        particles.push(new Particle(x, y, color));

      }

    }


    function gameOver() {

      gameState = "GAMEOVER";

      activeTarget = null;

    }


    function resetGame() {

      score = 0;

      wave = 1;

      enemies = [];

      lasers = [];

      particles = [];

      activeTarget = null;

      hudScore.textContent = score;

      startWave(wave);

      gameState = "PLAYING";

    }


    // ==========================================

    // 5. INPUT HANDLING

    // ==========================================

    window.addEventListener("keydown", (e) => {

      if (gameState === "START" || gameState === "GAMEOVER") {

        if (e.key === "Enter" || e.key === " ") resetGame();

        return;

      }


      const key = e.key.toLowerCase();

      if (!/^[a-z]$/.test(key)) return;


      // Acquire target if none is locked

      if (!activeTarget) {

        let bestTarget = null;

        let highestY = -Infinity;


        for (const enemy of enemies) {

          if (enemy.y > 0 && enemy.word[0].toLowerCase() === key) {

            if (enemy.y > highestY) {

              highestY = enemy.y;

              bestTarget = enemy;

            }

          }

        }

        activeTarget = bestTarget;

      }


      // Process input on current target

      if (activeTarget) {

        const expectedChar = activeTarget.word[activeTarget.typed].toLowerCase();

        if (key === expectedChar) {

          activeTarget.typed++;

          score += 10;

          hudScore.textContent = score;


          // Aim cannon and fire

          player.targetAngle = Math.atan2(activeTarget.y - player.y, activeTarget.x - player.x);

          lasers.push(new Laser(player.x, player.y, activeTarget.x, activeTarget.y));


          // Word completed

          if (activeTarget.typed >= activeTarget.word.length) {

            createExplosion(activeTarget.x, activeTarget.y, "#4ade80");

            enemies = enemies.filter(item => item !== activeTarget);

            activeTarget = null;

            score += 50;

            hudScore.textContent = score;


            // Trigger next wave if all enemies cleared

            if (enemies.length === 0) {

              wave++;

              startWave(wave);

            }

          }

        }

      }

    });


    // ==========================================

    // 6. MAIN LOOP

    // ==========================================

    function loop() {

      ctx.clearRect(0, 0, WIDTH, HEIGHT);


      if (gameState === "PLAYING") {

        // Updates

        lasers.forEach(l => l.update());

        lasers = lasers.filter(l => l.life > 0);


        particles.forEach(p => p.update());

        particles = particles.filter(p => p.life > 0);


        enemies.forEach(e => e.update());


        // Renders

        lasers.forEach(l => l.draw());

        enemies.forEach(e => e.draw());

        particles.forEach(p => p.draw());


        // Player Ship

        ctx.save();

        ctx.translate(player.x, player.y);

        ctx.rotate(player.targetAngle + Math.PI / 2);

        ctx.fillStyle = "#e0e7ff";

        ctx.beginPath();

        ctx.moveTo(0, -16);

        ctx.lineTo(-12, 12);

        ctx.lineTo(0, 6);

        ctx.lineTo(12, 12);

        ctx.closePath();

        ctx.fill();

        ctx.restore();

      } else {

        // Menu screens

        ctx.fillStyle = "#ffffff";

        ctx.textAlign = "center";

        ctx.font = "bold 26px sans-serif";


        if (gameState === "START") {

          ctx.fillText("TYPE TO SHOOT", WIDTH / 2, HEIGHT / 2 - 20);

          ctx.font = "16px sans-serif";

          ctx.fillStyle = "#94a3b8";

          ctx.fillText("Press SPACE to begin", WIDTH / 2, HEIGHT / 2 + 20);

        } else if (gameState === "GAMEOVER") {

          ctx.fillStyle = "#f43f5e";

          ctx.fillText("CRITICAL FAILURE", WIDTH / 2, HEIGHT / 2 - 20);

          ctx.font = "16px sans-serif";

          ctx.fillStyle = "#94a3b8";

          ctx.fillText(`Final Score: ${score} — Press SPACE to restart`, WIDTH / 2, HEIGHT / 2 + 20);

        }

      }


      requestAnimationFrame(loop);

    }


    loop();

  </script>
